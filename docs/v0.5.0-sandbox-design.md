# v0.5.0 Sandbox Isolation Verification - Technical Design

## Overview

Phase 3 introduces isolated test execution in Docker containers to eliminate false positives caused by environment inconsistencies, and adds performance regression detection to catch upgrades that degrade system performance.

**Target Release**: v0.5.0  
**Estimated Complexity**: High  
**Dependencies**: Docker, docker-compose (optional)

---

## 1. Problem Statement

### Current Limitations

1. **Environment Pollution**
   - Test failures may be caused by global npm/pip packages
   - Local development state (node_modules cache, .pyc files) affects results
   - Can't verify upgrades work in clean CI-like environment

2. **Database Testing Gap**
   - Many projects use databases (PostgreSQL, MySQL, MongoDB)
   - Current verification skips database-dependent tests
   - Real-world breakage often happens in data layer integration

3. **Performance Regression Blind Spot**
   - Upgrades may introduce performance degradation
   - No automated detection of slower builds, tests, or runtime
   - Users discover performance issues in production

### Success Criteria

- ✅ Run tests in isolated Docker container matching CI environment
- ✅ Support database services (PostgreSQL, MySQL, Redis, MongoDB)
- ✅ Detect performance regressions (build time, test time, bundle size)
- ✅ Graceful fallback when Docker unavailable
- ✅ Keep local environment clean (no side effects)

#### A. Sandbox Orchestrator (`src/engine/sandbox/orchestrator.ts`)

```typescript
export interface SandboxOptions {
  enabled: boolean;
  services?: string[];           // ['postgres', 'redis', 'mongodb']
  keepContainer?: boolean;       // Don't remove container after run
  dockerImage?: string;          // Custom base image
  timeout?: number;              // Max execution time (default: 600s)
}

export interface SandboxResult {
  passed: boolean;
  buildSuccess: boolean;
  testsPassed: boolean;
  services: ServiceHealth[];
  logs: string;
  performance: PerformanceMetrics;
  exitCode: number;
}

export async function runInSandbox(
  cwd: string,
  options: SandboxOptions
): Promise<SandboxResult> {
  // 1. Check Docker availability
  const dockerAvailable = await checkDockerAvailable();
  if (!dockerAvailable) {
    throw new Error('Docker is not available. Install Docker or run without --sandbox.');
  }

  // 2. Detect project type and required services
  const projectType = await detectProjectType(cwd);
  const services = options.services || await detectRequiredServices(cwd);

  // 3. Generate Dockerfile and docker-compose.yml
  const dockerfile = await generateDockerfile(projectType, cwd);
  const composeFile = await generateComposeFile(projectType, services);

  // 4. Write temporary files
  await writeFile(join(cwd, '.greenbump-Dockerfile'), dockerfile);
  if (services.length > 0) {
    await writeFile(join(cwd, '.greenbump-compose.yml'), composeFile);
  }

  try {
    // 5. Build container
    await buildDockerImage(cwd, 'greenbump-verify');

    // 6. Start services (if any)
    let composeProject: string | undefined;
    if (services.length > 0) {
      composeProject = await startComposeServices(cwd);
      await waitForServices(services, options.timeout);
    }

    // 7. Run verification in container
    const result = await runVerificationInContainer(cwd, {
      image: 'greenbump-verify',
      networkMode: composeProject ? `${composeProject}_default` : 'bridge',
      timeout: options.timeout,
    });

    // 8. Collect performance metrics
    const perfMetrics = await collectPerformanceMetrics(result.logs);

    return {
      passed: result.exitCode === 0,
      buildSuccess: result.buildSuccess,
      testsPassed: result.testsPassed,
      services: result.services,
      logs: result.logs,
      performance: perfMetrics,
      exitCode: result.exitCode,
    };
  } finally {
    // 9. Cleanup (unless --keep-container)
    if (!options.keepContainer) {
      await cleanupDocker(cwd);
    }
  }
}
```

export async function generateDockerfile(
  projectType: ProjectType,
  cwd: string
): Promise<string> {
  const templates: Record<ProjectType, string> = {
    nodejs: `
FROM node:20-slim

WORKDIR /workspace

# Copy package files
COPY package*.json ./
COPY pnpm-lock.yaml* ./
COPY yarn.lock* ./

# Install dependencies
RUN if [ -f pnpm-lock.yaml ]; then \\
      corepack enable && pnpm install --frozen-lockfile; \\
    elif [ -f yarn.lock ]; then \\
      yarn install --frozen-lockfile; \\
    else \\
      npm ci; \\
    fi

# Copy source code
COPY . .

# Run build if build script exists
RUN if grep -q '"build":' package.json; then npm run build; fi

# Default command: run tests
CMD ["npm", "test"]
`,
    python: `
FROM python:3.11-slim

WORKDIR /workspace

# Copy dependency files
COPY requirements*.txt ./
COPY pyproject.toml* ./
COPY setup.py* ./

# Install dependencies
RUN if [ -f requirements.txt ]; then \\
      pip install --no-cache-dir -r requirements.txt; \\
    elif [ -f pyproject.toml ]; then \\
      pip install --no-cache-dir -e .; \\
    fi

# Copy source code
COPY . .

# Default command: run tests
CMD ["pytest"]
`,
  };

  return templates[projectType] || templates.nodejs;
}

#### C. Performance Metrics (`src/engine/perf/metrics.ts`)

```typescript
export interface PerformanceMetrics {
  installTime?: number;      // seconds
  buildTime?: number;        // seconds
  testTime?: number;         // seconds
  bundleSize?: number;       // bytes
  memoryPeak?: number;       // MB
  timestamp: Date;
}

export interface PerformanceComparison {
  metric: string;
  before: number;
  after: number;
  delta: number;            // absolute change
  percentChange: number;    // percentage change
  isRegression: boolean;    // exceeded threshold
  severity: 'ok' | 'warning' | 'critical';
}

export async function captureBaseline(cwd: string): Promise<PerformanceMetrics> {
  console.log('📊 Capturing performance baseline...');
  
  const metrics: PerformanceMetrics = {
    timestamp: new Date(),
  };

  // Measure install time
  const installStart = Date.now();
  await execAsync('npm install', { cwd });
  metrics.installTime = (Date.now() - installStart) / 1000;

  // Measure build time (if build script exists)
  if (await hasBuildScript(cwd)) {
    const buildStart = Date.now();
    await execAsync('npm run build', { cwd });
    metrics.buildTime = (Date.now() - buildStart) / 1000;

    // Measure bundle size
    const distDir = join(cwd, 'dist');
    if (await exists(distDir)) {
      metrics.bundleSize = await getTotalSize(distDir);
    }
  }

  // Measure test time
  if (await hasTestScript(cwd)) {
    const testStart = Date.now();
    await execAsync('npm test', { cwd });
    metrics.testTime = (Date.now() - testStart) / 1000;
  }

  // Save baseline to .greenbump/perf-baseline.json
  await writeJSON(join(cwd, '.greenbump', 'perf-baseline.json'), metrics);
  
  return metrics;
}

export async function comparePerformance(
  baseline: PerformanceMetrics,
  current: PerformanceMetrics
): Promise<PerformanceComparison[]> {
  const thresholds = {
    installTime: 0.30,    // 30% slower
    buildTime: 0.20,      // 20% slower
    testTime: 0.30,       // 30% slower
    bundleSize: 0.15,     // 15% larger
    memoryPeak: 0.40,     // 40% more memory
  };

  const comparisons: PerformanceComparison[] = [];

  for (const [metric, threshold] of Object.entries(thresholds)) {
    const before = baseline[metric as keyof PerformanceMetrics] as number;
    const after = current[metric as keyof PerformanceMetrics] as number;

    if (before === undefined || after === undefined) continue;

    const delta = after - before;
    const percentChange = (delta / before) * 100;
    const isRegression = percentChange > threshold * 100;

    comparisons.push({
      metric,
      before,
      after,
      delta,
      percentChange,
      isRegression,
      severity: isRegression
        ? percentChange > threshold * 200
          ? 'critical'
          : 'warning'
        : 'ok',
    });
  }

  return comparisons;
}
```

#### D. Service Detector (`src/engine/sandbox/service-detector.ts`)

```typescript
export async function detectRequiredServices(cwd: string): Promise<string[]> {
  const services: Set<string> = new Set();

  // Check package.json dependencies
  const pkgPath = join(cwd, 'package.json');
  if (await exists(pkgPath)) {
    const pkg = await readJSON(pkgPath);
    const allDeps = {
      ...pkg.dependencies,
      ...pkg.devDependencies,
    };

    // Detect database drivers
    if (allDeps.pg || allDeps['pg-promise'] || allDeps.postgres) {
      services.add('postgres');
    }
    if (allDeps.mysql || allDeps.mysql2) {
      services.add('mysql');
    }
    if (allDeps.redis || allDeps.ioredis) {
      services.add('redis');
    }
    if (allDeps.mongodb || allDeps.mongoose) {
      services.add('mongodb');
    }
  }

  // Check docker-compose.yml
  const composeFiles = [
    'docker-compose.yml',
    'docker-compose.yaml',
    'docker-compose.test.yml',
  ];

  for (const file of composeFiles) {
    const composePath = join(cwd, file);
    if (await exists(composePath)) {
      const content = await readFile(composePath, 'utf-8');
      if (content.includes('postgres:')) services.add('postgres');
      if (content.includes('mysql:')) services.add('mysql');
      if (content.includes('redis:')) services.add('redis');
      if (content.includes('mongo:')) services.add('mongodb');
    }
  }

  // Check .env files for connection strings
  const envPath = join(cwd, '.env.test');
  if (await exists(envPath)) {
    const env = await readFile(envPath, 'utf-8');
    if (/postgres:\/\//.test(env)) services.add('postgres');
    if (/mysql:\/\//.test(env)) services.add('mysql');
    if (/redis:\/\//.test(env)) services.add('redis');
    if (/mongodb:\/\//.test(env)) services.add('mongodb');
  }

  return Array.from(services);
}
```

export async function checkDockerAvailable(): Promise<boolean> {
  try {
    await execAsync('docker --version');
    await execAsync('docker ps');
    return true;
  } catch {
    return false;
  }
}

// Fallback to local verification
if (!dockerAvailable && options.sandbox) {
  console.warn('⚠️  Docker not available, falling back to local verification');
  console.warn('   Install Docker for isolated testing: https://docs.docker.com/get-docker/');
  options.sandbox = false;
}

### 5.2 Service Startup Timeout

```typescript
export async function waitForServices(
  services: string[],
  timeout: number
): Promise<void> {
  const deadline = Date.now() + timeout * 1000;
  
  for (const service of services) {
    while (Date.now() < deadline) {
      const healthy = await checkServiceHealth(service);
      if (healthy) break;
      await sleep(1000);
    }
    
    if (Date.now() >= deadline) {
      throw new Error(`Timeout waiting for ${service} to become healthy`);
    }
  }
}
```

### 5.3 Resource Cleanup

```typescript
export async function cleanupDocker(cwd: string): Promise<void> {
  try {
    // Stop and remove compose services
    if (await exists(join(cwd, '.greenbump-compose.yml'))) {
      await execAsync('docker-compose -f .greenbump-compose.yml down -v', { cwd });
    }
    
    // Remove verification image
    await execAsync('docker rmi greenbump-verify', { cwd }).catch(() => {});
    
    // Remove temporary files
    await rm(join(cwd, '.greenbump-Dockerfile')).catch(() => {});
    await rm(join(cwd, '.greenbump-compose.yml')).catch(() => {});
  } catch (err) {
    console.warn('⚠️  Cleanup warning:', err);
  }
}
```

## 7. Performance Considerations

### Optimization Strategies

1. **Image Caching**
   - Reuse base images across runs
   - Cache npm install layer
   - Only rebuild when dependencies change

2. **Parallel Service Startup**
   - Start all services concurrently
   - Wait for health checks in parallel
   - Reduce total startup time

3. **Incremental Builds**
   - Mount Docker build cache
   - Use BuildKit for faster builds
   - Skip rebuild if Dockerfile unchanged

4. **Resource Limits**
   ```typescript
   const containerConfig = {
     memory: '2g',        // Limit memory
     cpus: '2.0',         // Limit CPU cores
     timeout: 600,        // 10 minute timeout
   };
   ```
